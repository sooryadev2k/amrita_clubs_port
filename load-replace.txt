async function loadClubs() {
    try {
        const snapshot = await db.collection('clubs').get();
        if (snapshot.empty) {
            console.log("Firestore is empty. Seeding dummy clubs...");
            clubs = getSeedClubs();
            for (let c of clubs) {
                try {
                    await db.collection('clubs').doc(c.id).set(c);
                } catch(err) {
                    console.warn("Could not save to Firestore (check rules):", err.message);
                }
            }
        } else {
            clubs = [];
            snapshot.forEach(doc => clubs.push(doc.data()));
        }
    } catch (e) {
        console.error("Firebase Error (falling back to local):", e.message);
        const stored = localStorage.getItem('amrita_clubs');
        clubs = stored ? JSON.parse(stored) : [];
        if (clubs.length === 0) {
            clubs = getSeedClubs();
        }
    }
    updateStats();
    renderCategoryFilters();
    renderClubsGrid();
    renderAdminTable();
}
